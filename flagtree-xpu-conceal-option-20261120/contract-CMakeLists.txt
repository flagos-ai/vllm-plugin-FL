cmake_minimum_required(VERSION 3.20)
project(conceal_option_contract LANGUAGES CXX)
include("${CMAKE_CURRENT_SOURCE_DIR}/${VARIANT}.cmake")
option(TRITON_XPU_CONCEAL_IR "Conceal XPU IR text in distributed builds" ON)
add_library(triton MODULE probe.cpp)
flagtree_configure_backend_triton_target(triton)
get_target_property(actual triton COMPILE_DEFINITIONS)
if(EXPECT_CONCEAL)
  if(NOT "TRITON_CONCEAL_IR=1" IN_LIST actual)
    message(FATAL_ERROR "Missing default XPU concealment: ${actual}")
  endif()
else()
  if("TRITON_CONCEAL_IR=1" IN_LIST actual)
    message(FATAL_ERROR "Unexpected concealment: ${actual}")
  endif()
endif()
file(WRITE "${CMAKE_BINARY_DIR}/actual-definitions.txt" "${actual}\n")
